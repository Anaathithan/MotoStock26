-- 1. Supplier Table
CREATE TABLE IF NOT EXISTS Supplier (
    supplierID INT AUTO_INCREMENT PRIMARY KEY,
    supplierName VARCHAR(100) NOT NULL,
    contact VARCHAR(20),
    address VARCHAR(200)
);

-- 2. PurchaseOrder Table
CREATE TABLE IF NOT EXISTS PurchaseOrder (
    poID INT AUTO_INCREMENT PRIMARY KEY,
    supplierID INT,
    orderDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    totalCost DECIMAL(10,2),
    status ENUM('Pending','Received','Partial') DEFAULT 'Pending',
    FOREIGN KEY (supplierID) REFERENCES Supplier(supplierID)
);

-- 3. PurchaseItem Table (for one-by-one entry)
CREATE TABLE IF NOT EXISTS PurchaseItem (
    purchaseItemID INT AUTO_INCREMENT PRIMARY KEY,
    poID INT,
    partName VARCHAR(100),
    quantity INT,
    boughtPrice DECIMAL(10,2),
    FOREIGN KEY (poID) REFERENCES PurchaseOrder(poID)
);

CREATE TABLE IF NOT EXISTS Sale (
    saleID INT AUTO_INCREMENT PRIMARY KEY,
    customerID INT,
    saleDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    subTotal DECIMAL(10,2),
    discountPercent DECIMAL(5,2) DEFAULT 0,
    grandTotal DECIMAL(10,2),
    paymentMethod ENUM('Cash','Online Transfer') DEFAULT 'Cash'
);

CREATE TABLE IF NOT EXISTS SaleItem (
    saleItemID INT AUTO_INCREMENT PRIMARY KEY,
    saleID INT,
    partName VARCHAR(100),
    quantity INT,
    sellingPrice DECIMAL(10,2),
    itemTotal DECIMAL(10,2),
    FOREIGN KEY (saleID) REFERENCES Sale(saleID)
);

CREATE TABLE `Notification` (
  `notificationID` int(11) NOT NULL AUTO_INCREMENT,
  `sentAt` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `type` varchar(50) NOT NULL,
  `title` varchar(255) NOT NULL,
  `message` text NOT NULL,
  `toEmail` varchar(255) DEFAULT NULL,
  `emailSent` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`notificationID`),
  KEY `idx_sentAt` (`sentAt`),
  KEY `idx_type` (`type`),
  KEY `idx_emailSent` (`emailSent`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
